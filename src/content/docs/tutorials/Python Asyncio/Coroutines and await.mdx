## async def creates a coroutine function

```python title="coroutine_def.py" showLineNumbers{1}
import asyncio


async def say_hi():
    await asyncio.sleep(0.1)
    return "hi"
```

## Calling vs awaiting

Calling a coroutine function returns a coroutine object.

```python title="calling_vs_awaiting.py" showLineNumbers{1}
import asyncio


async def say_hi():
    await asyncio.sleep(0.1)
    return "hi"


async def main():
    c = say_hi()          # coroutine object (NOT executed yet)
    msg = await say_hi()  # executed
    print(type(c))
    print(msg)


asyncio.run(main())
```

```p5 title="Calling a coroutine function does not run it" desc="A normal function call runs the body and returns a value. Calling an async def returns a coroutine OBJECT and runs nothing at all -- the body has not started. Only await drives it. This is the single most common asyncio mistake, and the giveaway is that your print statements never appear." height="360"
const STEPS = [
  { line: "def add(a, b): return a + b",   call: "add(2, 3)",     ran: true,  got: "5",            note: "a plain function runs immediately and hands back a value" },
  { line: "async def job(): print('hi')",  call: "job()",         ran: false, got: "<coroutine>",  note: "the body did NOT run -- nothing printed" },
  { line: "async def job(): print('hi')",  call: "job()",         ran: false, got: "<coroutine>",  note: "the object just sits there; discarding it warns 'never awaited'" },
  { line: "async def job(): print('hi')",  call: "await job()",   ran: true,  got: "None, after printing hi", note: "await is what actually drives the coroutine" },
];

let idx = 0;
let last = 0;

function setup() {
  createCanvas(620, 360);
  textFont("monospace");
}

function draw() {
  background(13, 17, 23);
  if (millis() - last > 2000) { last = millis(); idx = (idx + 1) % STEPS.length; }
  const s = STEPS[idx];

  noStroke();
  fill(150);
  textSize(11);
  textAlign(LEFT, TOP);
  text("definition", 20, 16);
  fill(205);
  textSize(13);
  text(s.line, 20, 36);

  fill(150);
  textSize(11);
  text("you write", 20, 74);
  fill(88, 166, 255);
  textSize(16);
  text(s.call, 20, 94);

  // did the body run?
  const y = 136;
  fill(24, 29, 36);
  rect(20, y, 270, 92, 6);
  fill(150);
  textSize(11);
  textAlign(LEFT, TOP);
  text("did the body run?", 32, y + 12);
  fill(s.ran ? color(140, 210, 160) : color(255, 123, 114));
  textSize(24);
  text(s.ran ? "YES" : "NO", 32, y + 34);
  fill(s.ran ? color(140, 210, 160) : color(255, 123, 114));
  textSize(10);
  text(s.ran ? "output appeared" : "no output at all", 32, y + 68);

  fill(24, 29, 36);
  rect(310, y, 290, 92, 6);
  fill(150);
  textSize(11);
  textAlign(LEFT, TOP);
  text("what you got back", 322, y + 12);
  fill(s.got.indexOf("coroutine") >= 0 ? color(219, 171, 90) : color(140, 210, 160));
  textSize(15);
  text(s.got, 322, y + 40);

  // the mental model
  fill(150);
  textSize(11);
  textAlign(LEFT, TOP);
  text("a coroutine object is a plan, not a result", 20, 252);

  const boxes = [
    { l: "async def", on: true },
    { l: "call it", on: idx >= 1 },
    { l: "coroutine object", on: idx >= 1 },
    { l: "await it", on: idx === 3 },
    { l: "it runs", on: idx === 3 },
  ];
  for (let i = 0; i < boxes.length; i++) {
    const x = 20 + i * 118;
    noStroke();
    fill(boxes[i].on ? color(36, 43, 52) : color(20, 25, 31));
    rect(x, 274, 108, 34, 5);
    fill(boxes[i].on ? color(205) : color(70, 78, 90));
    textSize(10);
    textAlign(CENTER, CENTER);
    text(boxes[i].l, x + 54, 291);
    if (i < boxes.length - 1) {
      fill(boxes[i + 1].on ? color(120) : color(50, 56, 64));
      text(">", x + 113, 291);
    }
  }

  fill(210, 200, 160);
  textSize(12);
  textAlign(LEFT, TOP);
  text(s.note, 20, 326);
}
```

## await points

You can only use `await` inside an `async def` function.

If you need to run async code from normal code, use:

- `asyncio.run(main())` (most common)

```p5 title="Every await is a place the loop can leave you" desc="A coroutine is not interrupted at arbitrary points. It runs straight through until it reaches an await, and only then does control go back to the event loop, which is free to run something else before coming back. That is why asyncio needs no locks for ordinary code between awaits -- and why a blocking call with no await freezes everything." height="380"
const CODE = [
  { txt: "async def fetch(name):", await_: false },
  { txt: "    print(name, 'start')", await_: false },
  { txt: "    data = await get(url)", await_: true },
  { txt: "    parsed = parse(data)", await_: false },
  { txt: "    await save(parsed)", await_: true },
  { txt: "    print(name, 'done')", await_: false },
];

// Two coroutines interleaving. Each entry: which line A and B are on.
const TL = [
  { a: 0, b: -1, who: "A", note: "A runs; B has not started" },
  { a: 1, b: -1, who: "A", note: "still A -- no await yet, so nothing can interrupt it" },
  { a: 2, b: -1, who: "A", note: "A hits await: control goes back to the loop" },
  { a: 2, b: 0, who: "B", note: "the loop starts B while A waits on I/O" },
  { a: 2, b: 1, who: "B", note: "B runs straight through -- no await yet" },
  { a: 2, b: 2, who: "B", note: "B hits await too; both are now suspended" },
  { a: 3, b: 2, who: "A", note: "A's I/O finished, so the loop resumes A" },
  { a: 4, b: 2, who: "A", note: "A reaches its second await and suspends again" },
  { a: 4, b: 3, who: "B", note: "B resumes where it left off" },
  { a: 5, b: 3, who: "A", note: "A finishes" },
];

let idx = 0;
let last = 0;

function setup() {
  createCanvas(620, 380);
  textFont("monospace");
}

function draw() {
  background(13, 17, 23);
  if (millis() - last > 1500) { last = millis(); idx = (idx + 1) % TL.length; }
  const s = TL[idx];

  noStroke();
  fill(200);
  textSize(12);
  textAlign(LEFT, TOP);
  text("two coroutines, one thread", 20, 14);

  // the code, with await lines marked
  for (let i = 0; i < CODE.length; i++) {
    const y = 44 + i * 26;
    const onA = s.a === i;
    const onB = s.b === i;
    noStroke();
    if (onA || onB) {
      fill(onA && s.who === "A" ? color(30, 48, 62) : onB && s.who === "B" ? color(52, 40, 24) : color(26, 31, 38));
      rect(20, y - 3, 380, 24, 4);
    }
    fill(CODE[i].await_ ? color(219, 171, 90) : color(160));
    textSize(12);
    textAlign(LEFT, TOP);
    text(CODE[i].txt, 30, y);
    if (CODE[i].await_) {
      fill(219, 171, 90);
      textSize(9);
      textAlign(LEFT, TOP);
      text("suspend point", 300, y + 2);
    }
    // markers
    if (onA) { fill(88, 166, 255); textAlign(RIGHT, TOP); textSize(11); text("A", 416, y); }
    if (onB) { fill(219, 171, 90); textAlign(RIGHT, TOP); textSize(11); text("B", 432, y); }
  }

  // who holds the thread
  fill(150);
  textSize(11);
  textAlign(LEFT, TOP);
  text("running right now", 450, 44);
  fill(s.who === "A" ? color(88, 166, 255) : color(219, 171, 90));
  textSize(34);
  text(s.who, 450, 64);

  fill(150);
  textSize(10);
  textAlign(LEFT, TOP);
  text("the other one is", 450, 112);
  fill(120);
  text("suspended at an", 450, 126);
  text("await, holding", 450, 140);
  text("nothing.", 450, 154);

  // the point
  stroke(50);
  line(20, 214, 600, 214);
  noStroke();
  fill(140, 210, 160);
  textSize(11);
  textAlign(LEFT, TOP);
  text("Between two awaits your code cannot be interrupted.", 20, 228);
  fill(150);
  textSize(10);
  text("That is why ordinary asyncio code needs no lock: there is one thread, and it only", 20, 248);
  text("changes hands at an await you can see in the source.", 20, 264);
  fill(255, 123, 114);
  textSize(10);
  text("The flip side: a blocking call with no await -- time.sleep, a sync DB driver, a heavy", 20, 290);
  text("loop -- freezes every other coroutine, because the loop never gets control back.", 20, 306);

  fill(210, 200, 160);
  textSize(12);
  textAlign(LEFT, TOP);
  text(s.note, 20, 336);
  fill(90, 100, 112);
  textSize(10);
  text("step " + (idx + 1) + " of " + TL.length, 20, 358);
}
```

## Common mistake

This does nothing useful:

```python title="wrong.py" showLineNumbers{1}
async def main():
    say_hi()  # forgot await
```

Because the coroutine is created but never awaited.
