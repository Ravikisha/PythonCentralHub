## Why TaskGroup?

`asyncio.gather` is useful, but it can make lifecycle and error handling messy.

`asyncio.TaskGroup` (Python 3.11+) provides **structured concurrency**:

- tasks are scoped to a block
- if one task fails, the group cancels the rest
- errors are grouped in an `ExceptionGroup`

## Example

```python title="taskgroup_basic.py" showLineNumbers{1}
import asyncio


async def job(name: str, delay: float):
    await asyncio.sleep(delay)
    return f"{name} done"


async def main():
    async with asyncio.TaskGroup() as tg:
        t1 = tg.create_task(job("A", 0.2))
        t2 = tg.create_task(job("B", 0.1))

    # task results are available after the block
    print(t1.result())
    print(t2.result())


asyncio.run(main())
```

## Handling failures

```python title="taskgroup_error.py" showLineNumbers{1}
import asyncio


async def ok():
    await asyncio.sleep(0.1)
    return 1


async def bad():
    await asyncio.sleep(0.05)
    raise ValueError("boom")


async def main():
    try:
        async with asyncio.TaskGroup() as tg:
            tg.create_task(ok())
            tg.create_task(bad())
    except* ValueError as eg:
        # ExceptionGroup support (PEP 654)
        print("handled:", eg)


asyncio.run(main())
```

```p5 title="When one task fails: gather leaves the others running, TaskGroup does not" desc="Three workers; the middle one raises. Both forms report the error, but they leave the program in different states. gather returns the exception and the surviving sibling carries on to completion -- checked afterwards, it reports done and not cancelled. TaskGroup cancels every sibling before it re-raises, wrapped in an ExceptionGroup. That is what structured concurrency means: the block does not finish while anything it started is still running." height="400"
const RUNS = {
  gather: [
    { t: 0.05, note: "w1 finishes normally",              state: ["done", "run", "run"] },
    { t: 0.10, note: "w2 raises -- gather re-raises it",  state: ["done", "fail", "run"] },
    { t: 0.15, note: "you catch the error here",          state: ["done", "fail", "run"] },
    { t: 0.30, note: "w6 is STILL RUNNING; nobody stopped it", state: ["done", "fail", "run"] },
    { t: 0.55, note: "w6 completes. done=True, cancelled=False", state: ["done", "fail", "done"] },
  ],
  taskgroup: [
    { t: 0.05, note: "w1 finishes normally",              state: ["done", "run", "run"] },
    { t: 0.10, note: "w2 raises",                         state: ["done", "fail", "run"] },
    { t: 0.11, note: "the group cancels every sibling at once", state: ["done", "fail", "cancel"] },
    { t: 0.12, note: "then re-raises as ExceptionGroup(1)", state: ["done", "fail", "cancel"] },
    { t: 0.12, note: "nothing the block started is still alive", state: ["done", "fail", "cancel"] },
  ],
};
const NAMES = ["w1 (fast)", "w2 (raises)", "w6 (slow)"];

let useTG = false;
let idx = 0;
let last = 0;

function setup() {
  createCanvas(620, 400);
  textFont("monospace");
}

function mousePressed() { useTG = !useTG; idx = 0; last = millis(); }

function toneFor(s) {
  if (s === "done") return [140, 210, 160];
  if (s === "fail") return [255, 123, 114];
  if (s === "cancel") return [219, 171, 90];
  return [88, 166, 255];
}

function draw() {
  background(13, 17, 23);
  const seq = useTG ? RUNS.taskgroup : RUNS.gather;
  if (millis() - last > 1700) { last = millis(); idx = (idx + 1) % seq.length; }
  const s = seq[idx];

  noStroke();
  fill(200);
  textSize(13);
  textAlign(LEFT, TOP);
  text(useTG ? "async with asyncio.TaskGroup() as tg:" : "await asyncio.gather(w1, w2, w6)", 20, 14);
  fill(90, 100, 112);
  textSize(10);
  text("click to switch", 20, 34);

  for (let i = 0; i < 3; i++) {
    const y = 66 + i * 56;
    const st = s.state[i];
    const t = toneFor(st);
    noStroke();
    fill(26, 31, 38);
    rect(20, y, 580, 42, 5);
    fill(t[0], t[1], t[2]);
    rect(20, y, 4, 42, 2, 0, 0, 2);

    fill(205);
    textSize(12);
    textAlign(LEFT, CENTER);
    text(NAMES[i], 38, y + 21);

    fill(t[0], t[1], t[2]);
    textSize(11);
    textAlign(RIGHT, CENTER);
    const label = st === "done" ? "completed" : st === "fail" ? "raised ValueError"
      : st === "cancel" ? "CANCELLED by the group" : "still running";
    text(label, 588, y + 21);

    // a small progress bar for the running ones
    if (st === "run") {
      fill(88, 166, 255, 120);
      rect(180, y + 30, Math.min(360, s.t * 700), 5, 2);
    }
  }

  fill(210, 200, 160);
  textSize(12);
  textAlign(LEFT, TOP);
  text(s.note, 20, 248);
  fill(90, 100, 112);
  textSize(10);
  text("t = " + s.t.toFixed(2) + " s", 520, 250);

  stroke(50);
  line(20, 276, 600, 276);
  noStroke();

  const rows = [
    { l: "gather", v: "sibling runs to completion", on: !useTG, c: [255, 123, 114] },
    { l: "TaskGroup", v: "sibling cancelled before the error escapes", on: useTG, c: [140, 210, 160] },
  ];
  for (let i = 0; i < 2; i++) {
    const y = 290 + i * 32;
    fill(rows[i].on ? color(32, 38, 46) : color(20, 25, 31));
    rect(20, y, 580, 26, 4);
    fill(rows[i].on ? color(rows[i].c[0], rows[i].c[1], rows[i].c[2]) : color(50, 56, 64));
    rect(20, y, 4, 26, 2, 0, 0, 2);
    fill(rows[i].on ? 210 : 90);
    textSize(11);
    textAlign(LEFT, CENTER);
    text(rows[i].l, 36, y + 13);
    fill(rows[i].on ? color(rows[i].c[0], rows[i].c[1], rows[i].c[2]) : color(60, 66, 74));
    textAlign(RIGHT, CENTER);
    text(rows[i].v, 588, y + 13);
  }

  fill(150);
  textSize(10);
  textAlign(LEFT, TOP);
  text("A task nobody is waiting for is a leak: it can hold a connection open, write after the", 20, 358);
  text("request is gone, or raise into a loop that has no one left to report to.", 20, 374);
}

- Requires Python 3.11+
- Consider TaskGroup for “start N tasks, all must finish” patterns
