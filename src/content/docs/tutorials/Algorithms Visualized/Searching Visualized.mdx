Finding a value in a list sounds trivial — until the list has a million items. How you
search matters enormously. This page contrasts the two classics: **linear search**, which
checks every item, and **binary search**, which is dramatically faster but needs a
**sorted** list.

## What you'll learn

- how linear search scans one-by-one (works on any list)
- how binary search halves the remaining range each step (needs sorted data)
- why binary search is *O(log n)* — a million items in ~20 checks

## Linear vs binary

```python title="search.py" showLineNumbers {12-14}
def linear_search(nums, target):
    for i, value in enumerate(nums):
        if value == target:
            return i          # found it
    return -1                 # not present

def binary_search(nums, target):   # nums MUST be sorted
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if nums[mid] == target:
            return mid
        elif nums[mid] < target:
            lo = mid + 1       # target is in the right half
        else:
            hi = mid - 1       # target is in the left half
    return -1

data = [2, 5, 8, 12, 16, 23, 38, 56, 72, 91]
print(binary_search(data, 23))    # 5
```

## Watch binary search

The blue range is what's still in play; the yellow bar is the **middle** being checked.
Each step throws away half the range. Click the canvas to pick a new target.

```p5 title="Binary search halves the range" desc="On a sorted list, binary search checks the middle and discards half the list each step. Click to search for a new target." height="240"
let arr = [];
let target = 0, lo = 0, hi = 0, mid = 0;
let found = false, done = false;
let t = 0;
function reset() {
  arr = [];
  let v = floor(random(2, 8));
  for (let k = 0; k < 12; k++) { arr.push(v); v += floor(random(2, 9)); }
  target = arr[floor(random(arr.length))];
  lo = 0; hi = arr.length - 1; mid = floor((lo + hi) / 2);
  found = false; done = false; t = 0;
}
function setup() {
  createCanvas(600, 240);
  textFont("monospace");
  textAlign(CENTER, CENTER);
  reset();
}
function draw() {
  background(13, 17, 23);
  const n = arr.length, bw = 44, gap = 6;
  const x0 = (width - (n * bw + (n - 1) * gap)) / 2, y = height / 2 + 10;
  for (let k = 0; k < n; k++) {
    const x = x0 + k * (bw + gap);
    const inRange = k >= lo && k <= hi;
    const isMid = k === mid && !done;
    noStroke();
    if (found && k === mid) fill(120, 200, 120);
    else if (isMid) fill(255, 211, 67);
    else if (inRange) fill(75, 139, 190);
    else fill(24, 30, 40);
    rect(x, y - 24, bw, 48, 8);
    fill(inRange || (found && k === mid) ? 255 : 80);
    textSize(15);
    text(arr[k], x + bw / 2, y);
  }
  fill(255, 211, 67);
  textSize(14);
  text("target = " + target, width / 2, y - 58);
  fill(107, 169, 221);
  textSize(12);
  text(done ? "found at index " + mid : "check middle: arr[" + mid + "] = " + arr[mid], width / 2, y + 46);
  t++;
  if (!done && t % 42 === 0) {
    if (arr[mid] === target) {
      found = true; done = true;
    } else {
      if (arr[mid] < target) lo = mid + 1;
      else hi = mid - 1;
      mid = floor((lo + hi) / 2);
    }
  }
  if (done && t % 150 === 0) reset();
}
function mousePressed() { reset(); }
```

Linear search may check all *n* items; binary search checks about *log₂ n*. For 1,000,000
sorted items that's the difference between up to a million comparisons and about **20**.
The catch: binary search only works if the list is already sorted.
