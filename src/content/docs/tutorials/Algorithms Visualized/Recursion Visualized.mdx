Recursion is a function that calls itself on a smaller version of the problem until it hits
a **base case**. It's elegant, but it's easy to write a recursive function that does an
astonishing amount of repeated work. The classic example is the Fibonacci sequence.

## What you'll learn

- how a recursive call branches into a **tree** of sub-calls
- why naive `fib(n)` is exponentially slow (it recomputes the same values)
- how memoization collapses that tree

## Naive Fibonacci

```python title="fib.py" showLineNumbers {2,4}
def fib(n):
    if n < 2:          # base case
        return n
    return fib(n - 1) + fib(n - 2)   # two recursive calls

print(fib(5))   # 5
```

Each call to `fib(n)` spawns **two** more calls. Draw that out and you get a tree — and the
same sub-problems (`fib(2)`, `fib(3)`) show up over and over:

```p5 title="The fib(5) call tree" desc="Every call branches into two. Green leaves are base cases. Notice how fib(3) and fib(2) get recomputed many times." height="320"
let nodes = [];
let leafX = 0;
let shown = 0;
let t = 0;
function build(nn, depth, parent) {
  const id = nodes.length;
  nodes.push({ n: nn, depth: depth, parent: parent, x: 0, children: [] });
  if (parent >= 0) nodes[parent].children.push(id);
  if (nn < 2) {
    nodes[id].x = leafX++;
  } else {
    build(nn - 1, depth + 1, id);
    build(nn - 2, depth + 1, id);
    const ch = nodes[id].children;
    nodes[id].x = (nodes[ch[0]].x + nodes[ch[1]].x) / 2;
  }
  return id;
}
function setup() {
  createCanvas(600, 320);
  textFont("monospace");
  textAlign(CENTER, CENTER);
  nodes = [];
  leafX = 0;
  build(5, 0, -1);
}
function nx(nd) { return map(nd.x, 0, leafX - 1, 40, width - 40); }
function ny(nd) { return 34 + nd.depth * 50; }
function draw() {
  background(13, 17, 23);
  const lim = min(shown, nodes.length);
  stroke(60, 70, 84);
  strokeWeight(1.5);
  for (let i = 0; i < lim; i++) {
    const nd = nodes[i];
    if (nd.parent >= 0) {
      const p = nodes[nd.parent];
      line(nx(nd), ny(nd), nx(p), ny(p));
    }
  }
  noStroke();
  for (let i = 0; i < lim; i++) {
    const nd = nodes[i];
    fill(nd.n < 2 ? color(60, 120, 90) : color(75, 139, 190));
    ellipse(nx(nd), ny(nd), 32, 32);
    fill(255);
    textSize(13);
    text(nd.n, nx(nd), ny(nd));
  }
  fill(255, 211, 67);
  textSize(12);
  textAlign(LEFT, CENTER);
  text(shown >= nodes.length ? nodes.length + " calls just to compute fib(5)!" : "expanding calls…", 18, height - 12);
  textAlign(CENTER, CENTER);
  t++;
  if (t % 13 === 0 && shown < nodes.length) shown++;
  if (shown >= nodes.length && t % 180 === 0) { shown = 0; t = 0; }
}
```

Fifteen calls for `fib(5)`; `fib(30)` makes over **2.6 million**. The fix is
**memoization** — cache each result so every sub-problem is computed once:

```python title="fib_memo.py"
from functools import lru_cache

@lru_cache
def fib(n):
    return n if n < 2 else fib(n - 1) + fib(n - 2)

print(fib(30))   # instant
```
