```python title="Parse date" showLineNumbers{1}

if "date" in df.columns:
    df["date"] = pd.to_datetime(df["date"], errors="coerce")
```

## Step 3: Trend plot

```python title="Trend" showLineNumbers{1}
import matplotlib.pyplot as plt

country = "India"
sub = df[df["location"] == country].sort_values("date")

plt.figure(figsize=(10, 4))
plt.plot(sub["date"], sub["new_cases"], label="new_cases")
plt.title(f"New cases over time - {country}")
plt.xlabel("Date")
plt.ylabel("New cases")
plt.xticks(rotation=20)
plt.tight_layout()
plt.show()
```

:::tip[From the book]
McKinney's US baby names chapter reveals real trends only after grouping by year
and looking at the shape over decades — a single year tells you almost nothing.
Daily case counts are the same: one noisy day means little. A rolling window
(`.rolling(7).mean()`) smooths the noise so the real trend underneath can show through.
:::

## Step 4: Moving average

```python title="Rolling average" showLineNumbers{1}
sub = sub.copy()
sub["new_cases_ma7"] = sub["new_cases"].rolling(7).mean()

plt.figure(figsize=(10, 4))
plt.plot(sub["date"], sub["new_cases"], alpha=0.4, label="daily")
plt.plot(sub["date"], sub["new_cases_ma7"], label="7-day avg")
plt.title(f"New cases (7-day avg) - {country}")
plt.legend()
plt.tight_layout()
plt.show()
```

## Step 5: Compare regions

```python title="Compare countries" showLineNumbers{1}
import seaborn as sns
```

## Visualize it

```p5 title="Daily cases vs 7-day average" desc="The faint line is noisy daily data; the bold amber line is the smoothed trend." height="260"
function setup() {
  createCanvas(600, 260);
  textFont("monospace");
  noLoop();
}
function draw() {
  background(13, 17, 23);
  const gx = 50, gy = 20, gw = 520, gh = 190;
  stroke(60, 70, 84);
  line(gx, gy + gh, gx + gw, gy + gh);
  line(gx, gy, gx, gy + gh);

  let seed = 3;
  function rnd() { seed = (seed * 9301 + 49297) % 233280; return seed / 233280; }
  const n = 40;
  const daily = [];
  const smooth = [];
  for (let i = 0; i < n; i++) {
    const trend = 0.2 + 0.5 * Math.sin(i / 12) * 0.5 + i / (n * 2.2);
    daily.push(Math.max(0.05, trend + (rnd() - 0.5) * 0.25));
  }
  for (let i = 0; i < n; i++) {
    let s = 0, c = 0;
    for (let k = Math.max(0, i - 6); k <= i; k++) { s += daily[k]; c++; }
    smooth.push(s / c);
  }
  noFill();
  stroke(75, 139, 190, 140);
  strokeWeight(1.2);
  beginShape();
  daily.forEach((v, i) => vertex(gx + (i / (n - 1)) * gw, gy + gh - v * gh));
  endShape();

  stroke(255, 211, 67);
  strokeWeight(2.5);
  beginShape();
  smooth.forEach((v, i) => vertex(gx + (i / (n - 1)) * gw, gy + gh - v * gh));
  endShape();

  noStroke();
  fill(205);
  textSize(12);
  textAlign(LEFT, TOP);
  text("blue = daily new cases   amber = 7-day average", gx, 6);
}
```
