```python title="ai_powered_customer_support_chatbot.py" showLineNumbers startLineNumber=34
class CLI:
    @staticmethod
    def run():
        print("AI-powered Customer Support Chatbot")
        bot = CustomerSupportChatbot()
        while True:
            cmd = input('> ')
            if cmd == 'exit':
                print("Goodbye!")
                break
            response = bot.respond(cmd)
            print(response)
```
