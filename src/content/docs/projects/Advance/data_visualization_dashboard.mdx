        plt.title('Data Visualization Dashboard')
        plt.xlabel('Category')
        plt.ylabel('Value')
        plt.savefig("data_visualization_dashboard.png", dpi=120, bbox_inches="tight")
        print("saved data_visualization_dashboard.png")
        plt.show()
