```python title="web_scraping_automation.py" showLineNumbers startLineNumber=4
class WebScrapingAutomation:
    def __init__(self):
        pass

    def scrape(self, url):
        response = requests.get(url)
        soup = BeautifulSoup(response.text, 'html.parser')
        print(f"Title of {url}: {soup.title.string}")
        return soup.title.string

    def demo(self):
        self.scrape('https://www.python.org')
```
