class CLI:
    @staticmethod
    def run():
        print("AI-powered Fraud Detection System")
        print("Commands: train <data_file>, predict <data_file>, exit")
        detector = FraudDetection()
        while True:
            cmd = input('> ')
            if cmd.startswith('train'):
                parts = cmd.split()
                if len(parts) < 2:
                    print("Usage: train <data_file>")
                    continue
                X = np.loadtxt(parts[1], delimiter=',')
                detector.train(X)
                print("Model trained.")
