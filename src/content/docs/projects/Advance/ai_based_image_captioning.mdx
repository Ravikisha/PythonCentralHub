class CLI:
    @staticmethod
    def run():
        print("AI-based Image Captioning")
        while True:
            cmd = input('> ')
            if cmd.startswith('train'):
                parts = cmd.split()
                if len(parts) < 3:
                    print("Usage: train <img_dir> <captions_file>")
                    continue
                img_dir, captions_file = parts[1], parts[2]
                cap = ImageCaptioner()
                cap.train(img_dir, captions_file)

                parts = cmd.split()
                if len(parts) < 2:
